cmake_minimum_required(VERSION 3.20)
project(f8sdk VERSION 0.1.0 LANGUAGES CXX)

set(F8SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)

option(F8_WITH_ZENOH "Build the Zenoh transport" ON)
option(F8SDK_BUILD_TESTS "Build SDK unit tests" ON)
option(F8SDK_BUILD_EXAMPLES "Build the minimal service example" OFF)
set(F8_PIXI_CPP_ENV_DIR "${F8SDK_ROOT}/.pixi/envs/default" CACHE PATH "Pixi dependency prefix")
list(APPEND CMAKE_PREFIX_PATH "${F8_PIXI_CPP_ENV_DIR}")
if(WIN32)
    list(APPEND CMAKE_PREFIX_PATH "${F8_PIXI_CPP_ENV_DIR}/Library")
endif()
if(CMAKE_SOURCE_DIR STREQUAL F8SDK_ROOT)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
    foreach(config IN LISTS CMAKE_CONFIGURATION_TYPES)
        string(TOUPPER "${config}" config_upper)
        set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
        set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib")
        set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib")
    endforeach()
endif()
find_package(Threads REQUIRED)
add_subdirectory(cpp)
if(F8SDK_BUILD_TESTS)
    add_subdirectory(tests)
endif()
if(F8SDK_BUILD_EXAMPLES)
    add_subdirectory(cpp/examples/minimal_service)
endif()
